## 1 承認待ちの中身を見る入り口を足す

`p12_api.py` に、パスの形と、`__call__` の分かれ道と、`show` を足します。

```python
REQUEST_PATH = re.compile(r"^/requests/(?P<id>[0-9a-f]{32})$")

# __call__ の分かれ道に足す（/approvals の行の後）
#     elif method == "GET" and REQUEST_PATH.match(path):
#         status, body = self.show(environ, REQUEST_PATH.match(path).group("id"))


def authorized(self, environ: dict[str, Any]) -> int:
    """承認の係の鍵を確かめる. 問題が無ければ 0、あれば HTTP の番号."""
    if not self.approver_token:
        return 403
    given = environ.get("HTTP_AUTHORIZATION", "").removeprefix("Bearer ")
    return 0 if hmac.compare_digest(given.encode("utf-8"), self.approver_token.encode("utf-8")) else 401


def show(self, environ: dict[str, Any], request_id: str) -> tuple[int, dict[str, Any]]:
    code = self.authorized(environ)
    if code:
        return code, {"error": "承認の係の鍵が要ります"}
    saved = self.store.load(request_id)
    if saved is None or saved[1].status != WAITING:
        return 404, {"error": "承認待ちの依頼が見つかりません"}
    pending = saved[1].pending[0]
    return 200, {"request_id": request_id, "approval": {"tool": pending["name"], "arguments": pending["arguments"]}}
```

`authorized` と `show` は `AgentApp` のメソッドとして足します（字下げを 1 段深くする）。`approve` の鍵の確かめも `authorized` に置き換えると、同じ確かめが 1 か所になります。

テスト（`test_p12_api.py` の `make_app`・`call` などを使う）:

```python
def test_show_pending_request(tmp_path: Path) -> None:
    app = make_app(tmp_path, [search("延長 回数"), file_issue()])
    _, body = call(app, "POST", "/ask", {"question": FILE})
    request_id = body["request_id"]
    assert call(app, "GET", f"/requests/{request_id}")[0] == 401
    status, shown = call(app, "GET", f"/requests/{request_id}", token=TOKEN)
    assert status == 200 and shown["approval"]["tool"] == "create_issue"
    assert call(app, "GET", "/requests/" + "0" * 32, token=TOKEN)[0] == 404
```

なぜこうするか:

- 承認の中身には、起票の本文（利用者の質問の内容）が入ります。承認の係だけが見られるように、承認の API と同じ鍵で守ります
- 承認待ちでない依頼は、無い依頼と同じ `404` にします。済んだ依頼の中身を、後から見られないようにするためです

## 2 却下したときのテスト

```python
def test_denied_approval_does_not_create_issue(tmp_path: Path) -> None:
    tracker = MemoryIssueTracker()
    app = make_app(tmp_path, [search("延長 回数"), file_issue(), say("起票は見送りました")], tracker)
    _, body = call(app, "POST", "/ask", {"question": FILE})
    decision = {"approved": False, "comment": "重複です"}
    status, body = call(app, "POST", f"/approvals/{body['request_id']}", decision, token=TOKEN)
    assert status == 200 and body["answer"] == "起票は見送りました"
    assert tracker.issues == []
    told = json.loads(app.make_chat().calls[-1][-1]["content"])  # モデルに渡った最後のメッセージ（ツールの結果）
    assert "承認しなかった" in told["error"] and told["comment"] == "重複です"
```

なぜこうするか:

- API を通しても、10-4 の「却下したら理由をモデルに伝える」が保たれていることを確かめます。API にしたことで、承認の扱いが変わっていないかの後退のテストです
- `make_app` の `make_chat` は、毎回同じ偽のモデルを返すので、テストの後から、モデルに渡ったメッセージを読めます

## 3 FastAPI の例に承認の API を足す（任意）

```python
import hmac
import os
import re
from pathlib import Path

from fastapi import FastAPI, Header, HTTPException
from pydantic import BaseModel

from roadmap_examples.p10_agent import Decision
from roadmap_examples.p11_guard import run_guarded
from roadmap_examples.p12_api import Fixed, StateStore

app = FastAPI()
STORE = StateStore(Path("var") / "pending")
TOOLS = []  # 12-1 の main.py と同じツール


def make_chat(): ...  # 12-1 の main.py と同じ


class ApprovalBody(BaseModel):
    approved: bool
    comment: str = ""


@app.post("/approvals/{request_id}")
def approve(request_id: str, body: ApprovalBody, authorization: str = Header(default="")) -> dict[str, str]:
    token = os.environ.get("APPROVER_TOKEN", "")
    if not token:
        raise HTTPException(status_code=403, detail="承認の API は無効です")
    given = authorization.removeprefix("Bearer ")
    if not hmac.compare_digest(given.encode("utf-8"), token.encode("utf-8")):
        raise HTTPException(status_code=401, detail="承認の係の鍵が違います")
    saved = STORE.load(request_id) if re.fullmatch(r"[0-9a-f]{32}", request_id) else None
    if saved is None:
        raise HTTPException(status_code=404, detail="承認待ちの依頼が見つかりません")
    question, state = saved
    result = run_guarded(make_chat(), question, TOOLS, Fixed(Decision(body.approved, body.comment)), state=state)
    STORE.delete(request_id)
    return {"status": result.status, "answer": result.answer}
```

なぜこう書くか:

- `Header` で受けた引数 `authorization` は、HTTP の `Authorization` ヘッダーに当たります（引数の名前の `_` は `-` に読み替えられます）
- `/ask` の側では、承認待ちのときに `STORE.save` して `202` を返すように直します（12-1 の WSGI の版の `run` と同じ）
- 状態のファイルを `request_id` で開くので、`request_id` の形（32 桁の 16 進数）を確かめてから開きます（WSGI の版はパスの形で確かめています）。
  確かめないと、`../` のような値で、置き場所の外のファイルを開かれる恐れがあります
